\section{Scope and State-of-the-Art Positioning}
Existing global food-affordability trackers evaluate national annual average prices against crude macronutrient targets ($N \times 6.25$). This dataset advances the empirical state of the art by integrating four layers into a single reproducible panel:
\begin{enumerate}
    \item \textbf{Bioavailable Amino-Acid Normalization (DIAAS \& 9-AA LP):} Harmonizes $4.45 \times 10^6$ retail and historical price quotes from crude nitrogen to human-usable indispensable amino acids using FAO Digestible Indispensable Amino Acid Scores (DIAAS) and solves a continuous 9-indispensable-amino-acid (9-AA) linear program at every spatial cell.
    \item \textbf{Subnational Spatial Friction \& Seasonality:} Maps retail protein costs onto $70{,}440$ hexagonal H3 Resolution-4 cells (plus Resolution-6 urban--rural gradients across $90$ metropolitan corridors ($959$ real outlets)), capturing last-mile transport wedges outside observed market hubs ($94.4\%$ of global population resides within $100\text{ km}$ of an observed hub) and lean-season price spikes.
    \item \textbf{Labor-Time Denominators \& Wage-Cliff Accounting:} Converts currency costs into minutes of work per $50\text{ g}$ DIAAS-adjusted protein across median and bottom-quintile ($\text{P20}$) wage distributions, identifying $3.21\text{ billion}$ people ($39.1\%$ of $8.20\text{ billion}$) facing a $\ge 25\%$ daily-wage cliff.
    \item \textbf{Multi-Century Nordhaus \& Gabrielli--von K\'arm\'an Panel (1700--2026):} Unifies $326\text{ years}$ of historical urban price-wage series, ethnographic labor-effort benchmarks, modern retail quotes, and $2026$ ex-ante techno-economic assessment (TEA) B2B ingredient benchmarks onto a unified biophysical frontier.
\end{enumerate}

\section{Formal Mathematical Specification}
\subsection{DIAAS-Adjusted Retail Unit Cost ($u_{i,c}$)}
For commodity $c$ in spatial cell $i$ with retail price $P_{i,c}$ ($\text{USD/kg}$), crude protein fraction $\rho_c$, and capped FAO score $\text{DIAAS}_c \in (0, 1.25]$:
\begin{equation}
u_{i,c} = \frac{P_{i,c}}{20 \cdot \rho_c \cdot \min(\text{DIAAS}_c,\, 1.0)}
\end{equation}
where $u_{i,c}$ denotes the retail cost per $50\text{ g}$ of bioavailable reference protein (daily adult requirement for a $62.5\text{ kg}$ adult at $0.8\text{ g/kg/day}$).

\subsection{Nine-Indispensable-Amino-Acid Linear Program ($\text{LP}_i$ vs.\ $\text{LP}_{v,i}$)}
At each cell $i$, the unconstrained minimum-cost protein basket $\text{LP}_i$ minimizes expenditure subject to meeting all nine FAO adult indispensable amino acid requirements $R_a$ ($a \in \{\text{His, Ile, Leu, Lys, SAA, AAA, Thr, Trp, Val}\}$ per $50\text{ g}$ reference protein):
\begin{equation}
\text{LP}_i = \min_{\mathbf{q} \ge \mathbf{0}} \sum_{c \in \mathcal{C}_i} P_{i,c}\, q_c \quad \text{s.t.} \quad \sum_{c \in \mathcal{C}_i} \alpha_{c,a}\, d_{c,a}\, q_c \ge R_a \;\;\forall a \in \{1,\dots,9\}
\end{equation}
Restricting $\mathcal{C}_i$ to plant-only foods ($\mathcal{C}_i^{\text{plant}}$) yields the plant-only complementarity shadow cost $\text{LP}_{v,i}$, satisfying the strict invariant $\text{LP}_{v,i} \ge \text{LP}_i$ across all $70{,}440$ Res-4 cells (global median complementarity premium $\text{LP}_{v,i}/\text{LP}_i = 1.180$, IQR: $1.179\text{--}1.180$).

\subsection{Labor-Time Cost ($L_{i,q}$) and Daily-Wage Share ($w_{i,q}$)}
Given an $8\text{-hour}$ ($480\text{-minute}$) workday and daily wage $W_{i,q}$ for income quantile $q \in \{\text{Median}, \text{P20}\}$:
\begin{equation}
w_{i,q} = \frac{u_i^*}{W_{i,q}} \times 100\%, \qquad L_{i,q} = 480 \cdot \frac{u_i^*}{W_{i,q}} \;\; (\text{minutes per 50g DIAAS protein})
\end{equation}
